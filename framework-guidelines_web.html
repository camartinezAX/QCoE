📊 Allure Reporting (v3)

This project uses Allure 3 (TypeScript-based, no Java required). The allure npm package is included as a devDependency and provides the CLI via npx allure.

💡
How It Works
  1. Clean previous results: npm run clean and npm run clean:state
  2. Run any test command (npm test, npm run bs:sanity, etc.) — results are saved to allure-results/
  3. Run npm run allure:generate to build the HTML report into allure-report/
  4. Run npm run allure:open to view it in the browser
  5. Or use npm run allure:serve as a one-step generate + open
⚠️
Always Clean Before Running Tests
Previous test results accumulate in allure-results/ and will pollute your report with stale data. Always run cleanup before executing tests:
bash 
# Clean reports and BrowserStack artifacts
npm run clean
npm run clean:state

# Then run tests (local or BrowserStack)
npm test
# or
npm run bs:sanity

# Then generate and view the report
npm run allure:serve
✅
Verify Installation
Run npx allure --version — you should see version 3.x.x. If you get an error, run npm install to restore dependencies.

🛠️ Utility Commands

Command Description

npm run allure:open Open Allure report in browser
npm run allure:serve Generate and open Allure report in one step

Page Objects (Functional Style)

💡
Functional Style
Use pure functions that receive page as parameter. No this.page or class instances.

✅ Correct (Functional)

// src/pages/login.page.js
const locators = require('../locators/login.locators');
const helpers = require('../utils/helpers');

async function login(page, { username, password, baseUrl }) {
  const url = baseUrl ? `${baseUrl}/login` : '/login';
  await page.goto(url);
  await helpers.inputText(page, locators.usernameInput, username);
  await helpers.inputText(page, locators.passwordInput, password);
  await helpers.clickElement(page, locators.signInButton);
  // ... rest of flow
  return true;
}

module.exports = { login };

❌ Wrong (Class-Based)

// DON'T do this
class LoginPage {
  constructor(page) {
    this.page = page;  // Avoid this pattern
  }
  
  async login(username, password) {
    await this.page.goto('/login');
    await this.page.fill('#user', username);
    // ...
  }
}

module.exports = LoginPage;

📝 Using in Tests

javascript 
const { test, expect } = require('../../src/fixtures/base.fixture');
const { login } = require('../../src/pages/login.page');
const Data = require('../../src/utils/data');

test.describe('Login @Login', () => {

      username: creds.Username,
      password: creds.Password,
      baseUrl: creds.BaseUrl  // URL from test data JSON
    });
  });
  
});

Locator Patterns

⚠️
XPath with Normalization
Always use translate(normalize-space(text()), ...) for case-insensitive, whitespace-normalized matching.
javascript 
// src/locators/login.locators.js
module.exports = {
  // ✅ Text-based (case-insensitive, whitespace-normalized)
  signInHeader: "//*[translate(normalize-space(text()), 'ABCDEFGHIJKLMNOPQRSTUVWXYZ', 'abcdefghijklmnopqrstuvwxyz') = 'sign into your account']",
  
  // ✅ Contains text (partial match)
  loadingIndicator: "//*[contains(translate(normalize-space(text()), 'ABCDEFGHIJKLMNOPQRSTUVWXYZ', 'abcdefghijklmnopqrstuvwxyz'), 'loading')]",
  
  // ✅ Simple index-based
  usernameInput: "(//input)[1]",
  passwordInput: "(//input)[2]",
  
  // ✅ Attribute-based
  submitButton: "//button[@type='submit']",
};

✅ DO - Use XPath

// Simple, readable XPath
"//*[translate(normalize-space(text())...]"
"(//input)[1]"
"//button[@type='submit']"
"//a[contains(@href, 'dashboard')]"

❌ DON'T - Complex CSS

// Avoid fragile selectors
"#app > div.main > form > input"
".btn-primary.submit-form.large"
"[data-testid='login-form'] input"

Code Quality Rules

✅ Use Helper Functions ▼

Always use helpers from src/utils/helpers.js for common operations.

✅ Correct

await helpers.waitForElement(page, locator);
await helpers.clickElement(page, locator);
await helpers.inputText(page, locator, text);
await helpers.validateElement(page, locator);

❌ Wrong

await page.waitForSelector(locator);
await page.click(locator);
await page.fill(locator, text);
await page.waitForTimeout(5000); // Never!
❌ No Commented Code ▼

Remove all commented code before committing. Version control tracks history. 

// ❌ DON'T leave commented code
async function loginFlow() {
  await helpers.click(submitBtn);
  // await helpers.wait(5000);
  // await oldMethod();
  // const debug = true;
}

// ✅ DO keep it clean
async function loginFlow() {
  await helpers.click(submitBtn);
}
❌ No Hard-Coded Waits ▼

Never use page.waitForTimeout(). Use smart waits instead.

✅ Smart Waits

await helpers.waitForElement(page, locator);
await helpers.waitForEnabledAndClick(page, btn);
await helpers.waitForElementToDisappear(page, loader);

❌ Hard Waits

await page.waitForTimeout(5000);
await new Promise(r => setTimeout(r, 3000));
// Just wasting time!
✅ Proper File Placement ▼

Files must be placed in the correct directory based on their purpose.

File Type Location
Page Objects src/pages/{feature}.page.js
Locators src/locators/{feature}.locators.js
Utilities src/utils/{utility}.js
Tests suites/{##}_{Feature}/{test}.spec.js
